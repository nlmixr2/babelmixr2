      ; Yeo-Johnson; the branches are picked with indicators because
      ; NM-TRAN cannot redefine IPRED in nested IF structures
      IPRED = RX_PRED_F_
      IPREDP = 0.0
      IF (IPRED .GE. 0.0) IPREDP = 1.0
      IPREDA = IPREDP*IPRED
      IPREDB = (1.0 - IPREDP)*IPRED
      IPREDL0 = 0.0
      IF (LAMBDA .EQ. 0.0) IPREDL0 = 1.0
      IPREDL2 = 0.0
      IF (LAMBDA .EQ. 2.0) IPREDL2 = 1.0
      IPREDY1 = ((IPREDA + 1.0)**LAMBDA - 1.0)/(LAMBDA + IPREDL0)
      IPREDY1 = IPREDL0*DLOG(IPREDA + 1.0) + (1.0 - IPREDL0)*IPREDY1
      IPREDY2 = (1.0 - (1.0 - IPREDB)**(2.0 - LAMBDA))/(2.0 - LAMBDA + IPREDL2)
      IPREDY2 = -IPREDL2*DLOG(1.0 - IPREDB) + (1.0 - IPREDL2)*IPREDY2
      IPRED = IPREDP*IPREDY1 + (1.0 - IPREDP)*IPREDY2
